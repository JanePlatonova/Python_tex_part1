	\section*{Алгоритм 1: Сума чисел від 0 до n}
	\textbf{Мета:} обчислити суму всіх цілих чисел від 0 до $n$ включно.
	\begin{tcolorbox}[colback=blue!10!white, colframe=blue!50!black, arc=2mm]
	\begin{algorithm}[H]
		\SetAlgoLined
		\KwIn{$n$}
		\KwOut{Сума чисел від $0$ до $n$}
		sum $\leftarrow$ 0\;
		\For{$i \leftarrow 0$ \KwTo $n$}{
			sum $\leftarrow$ sum + $i$\;
		}
		Вивести ``Сума чисел від 0 до'', $n$, ``дорівнює'', sum\;
		\caption{Сума\_від\_0\_до\_n}
	\end{algorithm}
	\end{tcolorbox}
	
	\textbf{Пояснення:}
	\begin{itemize}
		\item Змінна \texttt{sum} накопичує результат.
		\item Цикл проходить усі значення від 0 до $n$.
		\item На кожній ітерації додається поточне $i$ до суми.
	\end{itemize}
	
	\vspace{1em}
	
	\section*{Алгоритм 2: Сума $n$ чисел, введених користувачем}
	\textbf{Мета:} обчислити суму $n$ чисел, які користувач вводить з клавіатури.
	
	\begin{tcolorbox}[colback=blue!10!white, colframe=blue!50!black, arc=2mm]
	\begin{algorithm}[H]
		\SetAlgoLined
		\KwIn{$n$}
		\KwOut{Сума введених чисел}
		sum $\leftarrow$ 0\;
		\For{$i \leftarrow 1$ \KwTo $n$}{
			Ввести $x$\;
			sum $\leftarrow$ sum + $x$\;
		}
		Вивести ``Сума введених чисел дорівнює'', sum\;
		\caption{Сума\_n\_введених\_чисел}
	\end{algorithm}
\end{tcolorbox}
	
	\textbf{Пояснення:}
	\begin{itemize}
		\item Користувач вводить кожне число $x$.
		\item Цикл повторюється $n$ разів.
		\item Кожне введене число додається до загальної суми.
	\end{itemize}
	
	
	\begin{tcolorbox}[colback=blue!10!white, colframe=blue!50!black, arc=2mm]
		\begin{algorithm}[H]
			\SetAlgoLined
			\KwIn{$n$}
			\KwOut{Мінімальне з введених чисел}
			Ввести $x$\;
			min $\leftarrow$ $x$\;
			\For{$i \leftarrow 1$ \KwTo $n-1$}{
				Ввести $x$\;
				\If{$x < min$}{
					min $\leftarrow$ $x$\;
				}
			}
			Вивести ``Мінімальне число дорівнює'', min\;
			\caption{Пошук\_мінімального\_елемента}
		\end{algorithm}
	\end{tcolorbox}
	\textbf{Пояснення:}
	\begin{itemize}
		\item Перше введене число приймається за початковий мінімум.
		\item Цикл повторюється від другого до $n$-го числа.
		\item Якщо нове число менше за поточний мінімум, то мінімум оновлюється.
		\item Після завершення циклу виводиться знайдене мінімальне значення.
	\end{itemize}
